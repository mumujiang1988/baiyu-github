一、核心架构总览（最终定型）
1. 架构分层（严格解耦，四层闭环）
plaintext
【业务应用层】
├─ ERP通用模块（配置化单据、基础资料）
├─ 报表模块（复杂多表联查、统计分析）
└─ AI模块（动态查询、智能分析）
       ↓
【接入层（通用协议层）】
仅3个全局通用结构体，无任何业务相关类
├─ 通用查询结构体（UniversalQueryDTO）
├─ 通用保存结构体（UniversalSaveDTO）
└─ 通用分页返回体（UniversalPageVO）
       ↓
【中层双引擎工具集】（核心业务规则层）
├─ 通用CRUD引擎：解析JSON配置 → 自动生成预编译SQL（单表/简单业务）
└─ 报表模板引擎：解析SQL模板+片段占位符 → 生成动态安全SQL（多表/聚合）
       ↓
【公共能力层】（全局统一管控）
├─ 白名单校验（表/字段）
├─ SQL安全拦截（高危语句）
├─ 数据权限注入（适配RuoYi）
├─ 分页计算器（统一分页逻辑）
└─ 结果转换器（Map ↔ 通用返回格式）
       ↓
【底层内核】（私有封装，不对外暴露）
└─ JDBC+Map执行内核：仅负责SQL执行、结果集转Map、资源回收
       ↓
【数据源层】
多数据源 + 连接池 + 事务控制 → MySQL业务库/报表库
2. 核心设计理念（最终版）
底层统一：一套 JDBC+Map 内核，全局共用，杜绝重复开发，Map 仅在底层私有使用；
中层解耦：双引擎拆分，CRUD 适配简单业务，报表引擎适配复杂场景，互不干扰；
上层极简：舍弃所有业务 DTO/Entity，仅保留 3 个通用结构体，全程 JSON 配置驱动；
安全闭环：预编译参数 + 白名单 + SQL 拦截 + 数据权限，兼顾灵活与安全；
低代码落地：新增业务 / 报表 / 页面，仅需新增 JSON 配置，后端零代码。

核心职责：仅负责 SQL 执行、结果集转换、资源回收，不参与任何 SQL 拼装、业务逻辑，Map 仅内部使用，不对外暴露。

package com.ruoyi.common.db.core;

import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.jdbc.datasource.DataSourceUtils;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

import javax.sql.DataSource;
import java.sql.*;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * JDBC+Map底层执行内核（私有，仅给双引擎调用，不对外暴露）
 */
@Slf4j
@Component
public class JdbcMapCore {

    @Autowired
    private DataSource dataSource;

    /**
     * 通用查询：执行SQL，返回List<Map>（内部专用）
     */
    public List<Map<String, Object>> innerQuery(String sql, List<Object> params) {
        List<Map<String, Object>> result = new ArrayList<>();
        Connection conn = null;
        PreparedStatement pstmt = null;
        ResultSet rs = null;
        try {
            // 获取连接（复用Spring连接池，支持事务）
            conn = DataSourceUtils.getConnection(dataSource);
            pstmt = conn.prepareStatement(sql);
            // 绑定预编译参数（防注入）
            bindParams(pstmt, params);
            // 打印SQL日志（便于排错）
            log.info("JDBC执行查询SQL：{}，参数：{}", sql, params);
            // 执行查询，封装结果集为Map
            rs = pstmt.executeQuery();
            ResultSetMetaData meta = rs.getMetaData();
            int columnCount = meta.getColumnCount();
            while (rs.next()) {
                Map<String, Object> row = new HashMap<>();
                for (int i = 1; i <= columnCount; i++) {
                    String colName = meta.getColumnLabel(i);
                    row.put(colName, rs.getObject(i));
                }
                result.add(row);
            }
        } catch (SQLException e) {
            log.error("JDBC查询异常，SQL：{}，参数：{}", sql, params, e);
            throw new RuntimeException("数据库查询失败：" + e.getMessage());
        } finally {
            // 自动关闭资源，杜绝连接泄漏
            closeResources(conn, pstmt, rs);
        }
        return result;
    }

    /**
     * 通用增删改：执行SQL，返回影响行数
     */
    public int innerUpdate(String sql, List<Object> params) {
        Connection conn = null;
        PreparedStatement pstmt = null;
        try {
            conn = DataSourceUtils.getConnection(dataSource);
            pstmt = conn.prepareStatement(sql);
            bindParams(pstmt, params);
            log.info("JDBC执行增删改SQL：{}，参数：{}", sql, params);
            return pstmt.executeUpdate();
        } catch (SQLException e) {
            log.error("JDBC增删改异常，SQL：{}，参数：{}", sql, params, e);
            throw new RuntimeException("数据库操作失败：" + e.getMessage());
        } finally {
            closeResources(conn, pstmt, null);
        }
    }

    /**
     * 统计总数
     */
    public long innerCount(String sql, List<Object> params) {
        List<Map<String, Object>> result = innerQuery(sql, params);
        if (result.isEmpty()) {
            return 0;
        }
        Object count = result.get(0).values().iterator().next();
        return count instanceof Number ? ((Number) count).longValue() : 0;
    }

    /**
     * 批量执行增删改
     */
    @Transactional(rollbackFor = Exception.class)
    public int innerBatch(String sql, List<List<Object>> batchParams) {
        Connection conn = null;
        PreparedStatement pstmt = null;
        try {
            conn = DataSourceUtils.getConnection(dataSource);
            pstmt = conn.prepareStatement(sql);
            for (List<Object> params : batchParams) {
                bindParams(pstmt, params);
                pstmt.addBatch();
            }
            log.info("JDBC执行批量SQL：{}，批量参数：{}", sql, batchParams);
            int[] results = pstmt.executeBatch();
            // 统计总影响行数
            int total = 0;
            for (int res : results) {
                total += res;
            }
            return total;
        } catch (SQLException e) {
            log.error("JDBC批量操作异常，SQL：{}", sql, e);
            throw new RuntimeException("数据库批量操作失败：" + e.getMessage());
        } finally {
            closeResources(conn, pstmt, null);
        }
    }

    /**
     * 绑定预编译参数
     */
    private void bindParams(PreparedStatement pstmt, List<Object> params) throws SQLException {
        if (params == null || params.isEmpty()) {
            return;
        }
        for (int i = 0; i < params.size(); i++) {
            Object param = params.get(i);
            pstmt.setObject(i + 1, param);
        }
    }

    /**
     * 关闭JDBC资源
     */
    private void closeResources(Connection conn, PreparedStatement pstmt, ResultSet rs) {
        try {
            if (rs != null) {
                rs.close();
            }
            if (pstmt != null) {
                pstmt.close();
            }
            if (conn != null) {
                // 释放连接到连接池
                DataSourceUtils.releaseConnection(conn, dataSource);
            }
        } catch (SQLException e) {
            log.error("JDBC资源关闭异常", e);
        }
    }
}

package com.ruoyi.common.db.security;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.annotation.PostConstruct;
import java.util.HashSet;
import java.util.Set;

/**
 * 表/字段白名单校验器，防止非法表、敏感字段查询
 */
@Component
public class TableFieldWhiteList {

    /** 允许访问的表名（配置在application.yml中，用逗号分隔） */
    @Value("${db.white-list.tables:biz_,sys_}")
    private String tableWhiteList;

    /** 禁止访问的敏感字段 */
    private final Set<String> forbiddenFields = new HashSet<>();

    private Set<String> allowedTablePrefixes;

    @PostConstruct
    public void init() {
        // 初始化允许的表前缀（支持通配符，如biz_、sys_）
        allowedTablePrefixes = new HashSet<>();
        String[] tables = tableWhiteList.split(",");
        for (String table : tables) {
            allowedTablePrefixes.add(table.trim());
        }

        // 初始化禁止访问的敏感字段
        forbiddenFields.add("password");
        forbiddenFields.add("salt");
        forbiddenFields.add("secret");
        forbiddenFields.add("token");
    }

    /**
     * 校验表名是否合法
     */
    public boolean checkTable(String tableName) {
        if (tableName == null || tableName.isEmpty()) {
            return false;
        }
        // 校验表名前缀是否在白名单中
        return allowedTablePrefixes.stream().anyMatch(tableName::startsWith);
    }

    /**
     * 校验字段是否合法（排除敏感字段）
     */
    public boolean checkField(String fieldName) {
        if (fieldName == null || fieldName.isEmpty()) {
            return false;
        }
        return !forbiddenFields.contains(fieldName);
    }

    /**
     * 过滤非法字段
     */
    public List<String> filterFields(List<String> fields) {
        if (fields == null || fields.isEmpty()) {
            return new ArrayList<>();
        }
        List<String> validFields = new ArrayList<>();
        for (String field : fields) {
            if (checkField(field)) {
                validFields.add(field);
            }
        }
        return validFields;
    }
}

package com.ruoyi.common.db.security;

import org.springframework.stereotype.Component;

import java.util.Arrays;
import java.util.List;
import java.util.regex.Pattern;

/**
 * SQL安全拦截器，拦截高危SQL语句
 */
@Component
public class SqlSecurityInterceptor {

    /** 高危SQL关键字（不区分大小写） */
    private final List<String> highRiskKeywords = Arrays.asList(
            "drop", "alter", "truncate", "delete from", "update ", "insert into",
            "grant", "revoke", "create table", "drop table"
    );

    /** 全表删除/更新正则（无where条件） */
    private final Pattern fullTablePattern = Pattern.compile(
            "(delete from|update)\\s+\\w+\\s*where\\s*1=1|(delete from|update)\\s+\\w+\\s*$",
            Pattern.CASE_INSENSITIVE
    );

    /**
     * 校验SQL是否安全
     */
    public void checkSqlSafety(String sql) {
        if (sql == null || sql.isEmpty()) {
            throw new RuntimeException("SQL语句不能为空");
        }

        // 1. 拦截高危关键字
        String lowerSql = sql.toLowerCase();
        for (String keyword : highRiskKeywords) {
            if (lowerSql.contains(keyword.toLowerCase())) {
                throw new RuntimeException("禁止执行高危SQL操作，包含关键字：" + keyword);
            }
        }

        // 2. 拦截全表删除/更新（无where条件）
        if (fullTablePattern.matcher(sql).find()) {
            throw new RuntimeException("禁止执行全表删除/更新操作，请添加有效where条件");
        }
    }
}

package com.ruoyi.common.db.util;

import org.springframework.stereotype.Component;

/**
 * 分页计算器，统一分页偏移量计算、count语句生成
 */
@Component
public class PageCalculator {

    /**
     * 生成分页SQL（MySQL）
     */
    public String buildPageSql(String baseSql, int pageNum, int pageSize) {
        int offset = (pageNum - 1) * pageSize;
        return baseSql + " LIMIT " + offset + "," + pageSize;
    }

    /**
     * 生成count统计SQL
     */
    public String buildCountSql(String baseSql) {
        // 简化处理：替换select子句为count(1)，去掉order by
        String countSql = baseSql.replaceAll("^select.*?from", "select count(1) from");
        if (countSql.contains("order by")) {
            countSql = countSql.substring(0, countSql.indexOf("order by"));
        }
        return countSql;
    }
}

package com.ruoyi.common.db.engine.crud;

import com.ruoyi.common.db.core.JdbcMapCore;
import com.ruoyi.common.db.dto.UniversalQueryDTO;
import com.ruoyi.common.db.dto.UniversalSaveDTO;
import com.ruoyi.common.db.dto.vo.UniversalPageVO;
import com.ruoyi.common.db.security.DataScopeInjector;
import com.ruoyi.common.db.security.SqlSecurityInterceptor;
import com.ruoyi.common.db.security.TableFieldWhiteList;
import com.ruoyi.common.db.util.PageCalculator;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

/**
 * 通用CRUD引擎：结构化JSON配置 → 自动生成预编译SQL，适配单表/简单业务
 */
@Component
public class CommonCrudEngine {

    @Autowired
    private JdbcMapCore jdbcMapCore;
    @Autowired
    private TableFieldWhiteList tableFieldWhiteList;
    @Autowired
    private SqlSecurityInterceptor sqlSecurityInterceptor;
    @Autowired
    private DataScopeInjector dataScopeInjector;
    @Autowired
    private PageCalculator pageCalculator;

    /**
     * 分页查询（核心方法）
     */
    public UniversalPageVO pageQuery(UniversalQueryDTO dto) {
        // 1. 从配置表查询：表名、主键字段、字段配置（实际项目中需新增配置表，此处简化）
        String tableName = getTableNameByConfigCode(dto.getConfigCode());
        String pkField = getPkFieldByConfigCode(dto.getConfigCode());
        List<String> fields = getFieldsByConfigCode(dto.getConfigCode());

        // 2. 白名单校验
        if (!tableFieldWhiteList.checkTable(tableName)) {
            throw new RuntimeException("非法表名，不允许访问：" + tableName);
        }
        fields = tableFieldWhiteList.filterFields(fields);

        // 3. 拼接SQL（select + from + where + 数据权限 + order by）
        StringBuilder sql = new StringBuilder();
        List<Object> params = new ArrayList<>();

        // 拼接select字段
        sql.append("SELECT ").append(String.join(",", fields)).append(" FROM ").append(tableName).append(" WHERE 1=1 ");

        // 拼接查询条件
        buildWhereCondition(sql, params, dto.getConditions());

        // 拼接数据权限（开启则注入）
        if (dto.getEnableDataScope() != null && dto.getEnableDataScope()) {
            sql.append(dataScopeInjector.injectDataScope(tableName));
        }

        // 拼接排序
        buildOrderBy(sql, dto.getOrders());

        // 4. 安全校验
        sqlSecurityInterceptor.checkSqlSafety(sql.toString());

        // 5. 分页处理
        String pageSql = pageCalculator.buildPageSql(sql.toString(), dto.getPageNum(), dto.getPageSize());
        String countSql = pageCalculator.buildCountSql(sql.toString());

        // 6. 执行查询
        List<Map<String, Object>> rows = jdbcMapCore.innerQuery(pageSql, params);
        long total = jdbcMapCore.innerCount(countSql, params);

        // 7. 封装返回结果
        return UniversalPageVO.build(total, rows);
    }

    /**
     * 单表新增
     */
    public int insert(UniversalSaveDTO dto) {
        // 1. 获取配置信息
        String tableName = getTableNameByConfigCode(dto.getConfigCode());
        List<String> fields = getFieldsByConfigCode(dto.getConfigCode());

        // 2. 白名单校验
        if (!tableFieldWhiteList.checkTable(tableName)) {
            throw new RuntimeException("非法表名，不允许访问：" + tableName);
        }
        fields = tableFieldWhiteList.filterFields(fields);

        // 3. 拼接insert SQL
        StringBuilder sql = new StringBuilder();
        List<Object> params = new ArrayList<>();
        List<String> insertFields = new ArrayList<>();
        List<String> placeholders = new ArrayList<>();

        for (String field : fields) {
            if (dto.getFormData().containsKey(field)) {
                insertFields.add(field);
                placeholders.add("?");
                params.add(dto.getFormData().get(field));
            }
        }

        sql.append("INSERT INTO ").append(tableName)
                .append(" (").append(String.join(",", insertFields)).append(") ")
                .append("VALUES (").append(String.join(",", placeholders)).append(")");

        // 4. 安全校验
        sqlSecurityInterceptor.checkSqlSafety(sql.toString());

        // 5. 执行新增
        return jdbcMapCore.innerUpdate(sql.toString(), params);
    }

    /**
     * 单表编辑（按主键）
     */
    public int update(UniversalSaveDTO dto) {
        // 1. 获取配置信息
        String tableName = getTableNameByConfigCode(dto.getConfigCode());
        String pkField = getPkFieldByConfigCode(dto.getConfigCode());
        List<String> fields = getFieldsByConfigCode(dto.getConfigCode());

        // 2. 白名单校验
        if (!tableFieldWhiteList.checkTable(tableName)) {
            throw new RuntimeException("非法表名，不允许访问：" + tableName);
        }
        fields = tableFieldWhiteList.filterFields(fields);

        // 3. 校验主键
        Object pkValue = dto.getFormData().get(pkField);
        if (pkValue == null) {
            throw new RuntimeException("主键字段" + pkField + "不能为空");
        }

        // 4. 拼接update SQL
        StringBuilder sql = new StringBuilder();
        List<Object> params = new ArrayList<>();

        sql.append("UPDATE ").append(tableName).append(" SET ");
        for (String field : fields) {
            if (dto.getFormData().containsKey(field) && !field.equals(pkField)) {
                sql.append(field).append("=?, ");
                params.add(dto.getFormData().get(field));
            }
        }
        // 去掉最后一个逗号
        sql.deleteCharAt(sql.length() - 2);
        // 拼接主键条件
        sql.append(" WHERE ").append(pkField).append("=?");
        params.add(pkValue);

        // 5. 安全校验
        sqlSecurityInterceptor.checkSqlSafety(sql.toString());

        // 6. 执行更新
        return jdbcMapCore.innerUpdate(sql.toString(), params);
    }

    /**
     * 单表删除（按主键）
     */
    public int delete(String configCode, Object pkValue) {
        // 1. 获取配置信息
        String tableName = getTableNameByConfigCode(configCode);
        String pkField = getPkFieldByConfigCode(configCode);

        // 2. 白名单校验
        if (!tableFieldWhiteList.checkTable(tableName)) {
            throw new RuntimeException("非法表名，不允许访问：" + tableName);
        }

        // 3. 拼接delete SQL
        String sql = "DELETE FROM " + tableName + " WHERE " + pkField + "=?";
        List<Object> params = new ArrayList<>();
        params.add(pkValue);

        // 4. 安全校验
        sqlSecurityInterceptor.checkSqlSafety(sql.toString());

        // 5. 执行删除
        return jdbcMapCore.innerUpdate(sql.toString(), params);
    }

    /**
     * 拼接查询条件（eq/like/in/between等）
     */
    private void buildWhereCondition(StringBuilder sql, List<Object> params, List<UniversalQueryDTO.ConditionItem> conditions) {
        if (conditions == null || conditions.isEmpty()) {
            return;
        }

        for (UniversalQueryDTO.ConditionItem item : conditions) {
            // 校验字段合法性
            if (!tableFieldWhiteList.checkField(item.getField())) {
                continue;
            }

            String op = item.getOp();
            Object value = item.getValue();
            if (value == null) {
                continue;
            }

            switch (op.toLowerCase()) {
                case "eq":
                    sql.append(" AND ").append(item.getField()).append("=?");
                    params.add(value);
                    break;
                case "like":
                    sql.append(" AND ").append(item.getField()).append(" LIKE ?");
                    params.add("%" + value + "%");
                    break;
                case "in":
                    if (value instanceof List) {
                        List<?> inValues = (List<?>) value;
                        String inPlaceholder = String.join(",", new String[inValues.size()]);
                        sql.append(" AND ").append(item.getField()).append(" IN (").append(inPlaceholder).append(")");
                        params.addAll(inValues);
                    }
                    break;
                case "between":
                    if (value instanceof List && ((List<?>) value).size() == 2) {
                        List<?> betweenValues = (List<?>) value;
                        sql.append(" AND ").append(item.getField()).append(" BETWEEN ? AND ?");
                        params.add(betweenValues.get(0));
                        params.add(betweenValues.get(1));
                    }
                    break;
                // 可扩展gt/lt/ge/le等运算符
            }
        }
    }

    /**
     * 拼接排序条件
     */
    private void buildOrderBy(StringBuilder sql, List<UniversalQueryDTO.OrderItem> orders) {
        if (orders == null || orders.isEmpty()) {
            return;
        }

        sql.append(" ORDER BY ");
        for (int i = 0; i < orders.size(); i++) {
            UniversalQueryDTO.OrderItem item = orders.get(i);
            if (tableFieldWhiteList.checkField(item.getField())) {
                sql.append(item.getField()).append(" ").append(item.getSort().toUpperCase());
                if (i != orders.size() - 1) {
                    sql.append(", ");
                }
            }
        }
    }
