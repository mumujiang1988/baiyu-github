-- =============================================
-- 单据关系全链路查询（自动读取配置 + 构建关系 + 检查生成状态）
-- 使用方式：只需替换 👇 下面的 '当前单据表名' 为你打开的单据
-- =============================================
WITH RECURSIVE
-- 1. 读取配置表，构建完整关系链
bill_relation AS (
    SELECT 
        id,
        source,
        goal,
        business_name,
        remark
    FROM t_identifier_dictionary
    WHERE source = '当前单据表名'  -- 👈 替换成你当前打开的单据表名
    
    UNION ALL
    
    -- 递归向上找所有上游
    SELECT 
        t.id, t.source, t.goal, t.business_name, t.remark
    FROM t_identifier_dictionary t
    JOIN bill_relation br ON br.source = t.goal
    
    UNION ALL
    
    -- 递归向下找所有下游
    SELECT 
        t.id, t.source, t.goal, t.business_name, t.remark
    FROM t_identifier_dictionary t
    JOIN bill_relation br ON br.goal = t.source
),

-- 2. 去重 + 排序
distinct_chain AS (
    SELECT DISTINCT
        source AS table_code,
        business_name AS table_name
    FROM bill_relation
    ORDER BY id
),

-- 3. 动态判断单据是否生成（核心）
final_result AS (
    SELECT
        table_code,
        table_name,
        (
            CASE
                WHEN table_code = 't_sale_xsddxx' THEN 
                    (SELECT COUNT(*) > 0 FROM t_sale_xsddxx)
                WHEN table_code = 't_sale_order' THEN 
                    (SELECT COUNT(*) > 0 FROM t_sale_order)
                WHEN table_code = 'po_order_bill_head' THEN 
                    (SELECT COUNT(*) > 0 FROM po_order_bill_head)
                WHEN table_code = 'receive_notice' THEN 
                    (SELECT COUNT(*) > 0 FROM receive_notice)
                WHEN table_code = 'purchase_instock' THEN 
                    (SELECT COUNT(*) > 0 FROM purchase_instock)
                WHEN table_code = 'f_scp_payable' THEN 
                    (SELECT COUNT(*) > 0 FROM f_scp_payable)
                WHEN table_code = 'delivery_notice' THEN 
                    (SELECT COUNT(*) > 0 FROM delivery_notice)
                WHEN table_code = 'f_sal_outbound' THEN 
                    (SELECT COUNT(*) > 0 FROM f_sal_outbound)
                WHEN table_code = 't_ar_receivable' THEN 
                    (SELECT COUNT(*) > 0 FROM t_ar_receivable)
                ELSE 0
            END
        ) AS is_generated
    FROM distinct_chain
)

-- 最终输出：关系图列表
SELECT
    table_code AS '单据表名',
    table_name AS '单据显示名',
    CASE WHEN is_generated = 1 THEN '已生成' ELSE '未生成' END AS '状态'
FROM final_result
ORDER BY field(table_code,
    't_sale_xsddxx',
    't_sale_order',
    'po_order_bill_head',
    'receive_notice',
    'purchase_instock',
    'f_scp_payable',
    'delivery_notice',
    'f_sal_outbound',
    't_ar_receivable'
);
